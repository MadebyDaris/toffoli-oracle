cmake_minimum_required(VERSION 3.16)
project(toffoli_oracle LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# mockturtle is used unmodified, from a checkout in third_party/ (see README.md, Quick start)
set(MOCKTURTLE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/mockturtle CACHE PATH "mockturtle source tree")
if(NOT EXISTS ${MOCKTURTLE_DIR}/CMakeLists.txt)
  message(FATAL_ERROR "mockturtle not found at ${MOCKTURTLE_DIR}; run: git clone https://github.com/lsils/mockturtle.git third_party/mockturtle")
endif()
set(MOCKTURTLE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MOCKTURTLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MOCKTURTLE_BUILD_EXPERIMENTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${MOCKTURTLE_DIR} third_party/mockturtle EXCLUDE_FROM_ALL)

add_subdirectory(opt)
